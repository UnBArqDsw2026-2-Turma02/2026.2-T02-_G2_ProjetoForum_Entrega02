## Subgrupo 1

- **Ata de Reunião:** Como iniciativa extra para apoiar a rastreabilidade e o acompanhamento das decisões arquiteturais tomadas durante as sessões de modelagem, o subgrupo elaborou uma [Ata de Reunião — Arquitetura DSW (15/09)](iniciativas_extras/atas_reuniao/subequipe_01/ata-15-09.md). Esta ata detalha as atividades e materiais desenvolvidos por Laura, Caio, Artur e Zarbielli, incluindo materiais que ainda não foram adicionados aos relatórios principais.
- **Modelo Lógico (DDL) e Prototipação Arquitetural:** Além dos diagramas UML exigidos pela entrega, a equipe converteu a modelagem estática (MVC) para uma prova de conceito em código usando FastAPI e SQLAlchemy. A partir dessa prototipação técnica, validou-se com sucesso a estratégia de herança de banco de dados (*Joined Table Inheritance*) para a classe abstrata `Conteudo` e suas derivadas (`Pergunta` e `Resposta`). O resultado dessa validação gerou um modelo lógico de banco de dados em formato DDL (MySQL), evidenciando a robustez da arquitetura desenhada contra as regras de negócio.

```sql
-- Script de Criação do Banco de Dados em MySQL (Subequipe 1 - Conteúdo)
-- Gerado com base no mapeamento MVC (Joined Table Inheritance)

CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    senha_hash VARCHAR(255) NOT NULL
);

CREATE TABLE conteudos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    corpo TEXT NOT NULL,
    data_criacao DATETIME DEFAULT CURRENT_TIMESTAMP,
    autor_id INT,
    tipo_conteudo VARCHAR(50),
    FOREIGN KEY (autor_id) REFERENCES usuarios(id)
);

CREATE TABLE perguntas (
    id INT PRIMARY KEY,
    titulo VARCHAR(255) NOT NULL,
    status VARCHAR(50) DEFAULT 'ABERTA',
    FOREIGN KEY (id) REFERENCES conteudos(id) ON DELETE CASCADE
);

CREATE TABLE respostas (
    id INT PRIMARY KEY,
    aceita BOOLEAN DEFAULT FALSE,
    pergunta_id INT NOT NULL,
    FOREIGN KEY (id) REFERENCES conteudos(id) ON DELETE CASCADE,
    FOREIGN KEY (pergunta_id) REFERENCES perguntas(id) ON DELETE CASCADE
);

CREATE TABLE tags (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE pergunta_tag (
    pergunta_id INT NOT NULL,
    tag_id INT NOT NULL,
    PRIMARY KEY (pergunta_id, tag_id),
    FOREIGN KEY (pergunta_id) REFERENCES perguntas(id) ON DELETE CASCADE,
    FOREIGN KEY (tag_id) REFERENCES tags(id) ON DELETE CASCADE
);

CREATE TABLE edicoes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    corpo_anterior TEXT NOT NULL,
    data_modificacao DATETIME DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(50) DEFAULT 'PENDENTE',
    conteudo_id INT NOT NULL,
    FOREIGN KEY (conteudo_id) REFERENCES conteudos(id) ON DELETE CASCADE
);

CREATE TABLE moderacoes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    status VARCHAR(50) NOT NULL,
    data_moderacao DATETIME DEFAULT CURRENT_TIMESTAMP,
    moderador_id INT NOT NULL,
    conteudo_id INT NOT NULL,
    edicao_id INT NULL,
    FOREIGN KEY (moderador_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (conteudo_id) REFERENCES conteudos(id) ON DELETE CASCADE,
    FOREIGN KEY (edicao_id) REFERENCES edicoes(id) ON DELETE SET NULL
);
```
